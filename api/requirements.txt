# ========== WEB FRAMEWORK ==========
# FastAPI: framework web moderno con validación automática
# uvicorn: servidor ASGI (similar a gunicorn pero para async/await)
fastapi==0.115.0
uvicorn[standard]==0.30.6

# ========== DATABASE ==========
# SQLAlchemy: ORM (Object-Relational Mapping)
# [asyncio]: soporte para operaciones asincrónicas
# asyncpg: driver asincrónico para PostgreSQL (más rápido que psycopg2)
# alembic: sistema de migraciones para cambios en el esquema
sqlalchemy[asyncio]==2.0.32
asyncpg==0.29.0
alembic==1.13.2

# ========== REALTIME / EVENTS ==========
# redis: cliente para conectar a Redis (pub/sub, cache)
redis==5.0.8

# ========== CONFIGURATION ==========
# pydantic-settings: carga variables de entorno y valida tipos
# email-validator: valida direcciones de email (usado por pydantic)
pydantic-settings==2.4.0
email-validator==2.2.0

# ========== AUTHENTICATION ==========
# python-jose: genera y valida JWT tokens
# [cryptography]: backend criptográfico para firmas
# passlib: librería de hashing (bcrypt, argon2, etc.)
# [bcrypt]: algoritmo específico de hashing (muy seguro)
python-jose[cryptography]==3.3.0
passlib[bcrypt]==1.7.4

# ========== TESTING ==========
# pytest: framework de tests (ejecuta y reporta)
# pytest-asyncio: soporte para tests asincronos
# httpx: cliente HTTP similar a requests pero async
pytest==8.3.3
pytest-asyncio==0.24.0
httpx==0.27.2
